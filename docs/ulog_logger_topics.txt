# PX4 custom logger topic list for bench thrust/torque testing.
#
# Copy to the Pixhawk SD card as:  /fs/microsd/etc/logging/logger_topics.txt
# then reboot the flight controller. Format is one topic per line:
#
#     <topic_name>, <interval_ms>[, <instance>]
#
# Providing this file REPLACES the default topic list, so everything wanted must
# be listed. Verify against your firmware's docs before relying on it -- the
# path and format have been stable for years but are not guaranteed.
#
# Why this is worth doing (measured on log_3_2026-7-24-00-27-40.ulg):
#   - actuator_outputs logs at 100 ms (10 Hz) and battery_status at 200 ms (5 Hz)
#     by default. That is a configuration choice, NOT a limit: sensor_combined
#     logs at 220 Hz in the same file, so the logger and SD card have ~20x more
#     headroom than these two topics use.
#   - ~30% of the log volume is topics a static bench cannot use at all
#     (estimator_*, navigator_*, rtl_*, position_setpoint_*, transponder_*).
#     Dropping them shrinks the log while raising the rates that matter.

# --- what the thrust/torque map is actually built from ---
actuator_outputs, 20            # 50 Hz: measured PWM per channel, matches the load cell
battery_status, 50              # 20 Hz: voltage/current, and no-load voltage between runs
actuator_motors, 20             # 50 Hz: normalized motor commands
esc_status, 50                  # 20 Hz: RPM, if the ESCs support telemetry

# --- useful context, cheap ---
vehicle_command, 0              # every ACTUATOR_TEST the Pi sends (0 = unlimited)
vehicle_command_ack, 0          # and whether PX4 accepted it
system_power, 200               # 5 Hz: rail voltages, catches brownouts
cpuload, 1000
vehicle_status, 200

# --- vibration: explains the load-cell noise floor if you ever want to chase it ---
# sensor_combined, 5            # 200 Hz. Uncomment only when investigating noise;
                                # this one topic is ~25% of the log volume.
sensor_accel, 20                # 50 Hz, lighter alternative for vibration level

# --- deliberately NOT logged on a static bench ---
# estimator_*            EKF has no meaningful solution bolted to a table
# navigator_*, rtl_*     no mission
# position_setpoint_*    no position control
# transponder_report     no ADS-B
# vehicle_local_position 55 fields of nothing useful here (6% of volume)
